function locateRing(region,circle,expectedX,expectedY){
  log(60,1);const {width,height,data}=region.imageData,mask=new Uint8Array(width*height),queue=new Int32Array(width*height),redPixel=index=>data[index]>=190&&data[index+1]<=135&&data[index+2]<=150&&data[index]-data[index+1]>=70&&data[index]-data[index+2]>=55&&data[index+3]>=120;let redCount=0;
  for(let pixel=0,index=0;pixel<mask.length;pixel++,index+=4)if(redPixel(index)){mask[pixel]=1;redCount++}const expectedLocalX=expectedX-region.x,expectedLocalY=expectedY-region.y,diameter=circle.radius*2,candidates=[];log(60,2);
  for(let seed=0;seed<mask.length;seed++){if(mask[seed]!==1)continue;let head=0,tail=0,count=0,minX=width,maxX=-1,minY=height,maxY=-1;queue[tail++]=seed;mask[seed]=2;
    while(head<tail){const pixel=queue[head++],x=pixel%width,y=(pixel/width)|0;count++;if(x<minX)minX=x;if(x>maxX)maxX=x;if(y<minY)minY=y;if(y>maxY)maxY=y;const neighbours=[pixel-1,pixel+1,pixel-width,pixel+width];for(let n=0;n<4;n++){const next=neighbours[n];if(next<0||next>=mask.length||mask[next]!==1)continue;if((n===0&&x===0)||(n===1&&x===width-1))continue;mask[next]=2;queue[tail++]=next}}
    const boxWidth=maxX-minX+1,boxHeight=maxY-minY+1;if(count<Math.max(24,circle.radius*2)||boxWidth<circle.radius||boxHeight<circle.radius)continue;const centerX=(minX+maxX)/2,centerY=(minY+maxY)/2,sizeError=(Math.abs(boxWidth-diameter)+Math.abs(boxHeight-diameter))/diameter,roundError=Math.abs(boxWidth-boxHeight)/diameter,distance=Math.hypot(centerX-expectedLocalX,centerY-expectedLocalY)/Math.max(circle.radius,1),density=count/(boxWidth*boxHeight),densityError=Math.abs(density-0.16)/0.16,score=1-(sizeError*0.34+roundError*0.18+Math.min(distance,4)*0.18+Math.min(densityError,3)*0.12);candidates.push({centerX,centerY,boxWidth,boxHeight,count,score})
  }
  candidates.sort((a,b)=>b.score-a.score);const best=candidates[0];if(!best||best.score<0.30){log(60,3);return null}const located={x:region.x+best.centerX,y:region.y+best.centerY,radius:circle.radius,visualRadius:(best.boxWidth+best.boxHeight)/4,confidence:Math.max(0,Math.min(1,best.score)),redPixels:best.count,redCount};log(60,4);return located;
